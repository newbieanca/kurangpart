// Pending Part Recehan - Apps Script // Layout tab DATA (baris 1 = header, data mulai baris 2): // A=ID B=Tanggal C=Nama D=Line E=Model F=Part G=Kurang_Awal H=Datang_CMC I=Sisa // J=Status_Awal K=Remarks L=Update_CMC M=Status_CMC N=Terakhir_Update O=Update_Oleh var TZ = 'Asia/Jakarta'; // Isi SS_ID hanya jika script dibuat terpisah (bukan dari Extensions > Apps Script di dalam Sheet). // ID ada di URL Sheet: docs.google.com/spreadsheets/d//edit var SS_ID = ''; function sh_(n) { var ss = SS_ID ? SpreadsheetApp.openById(SS_ID) : SpreadsheetApp.getActiveSpreadsheet(); if (!ss) throw new Error('Script belum terhubung ke Sheet. Isi SS_ID di atas file.'); var sh = ss.getSheetByName(n); if (!sh) throw new Error('Tab "' + n + '" tidak ditemukan di Sheet.'); return sh; } function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); } function fmt_(v, p) { if (v instanceof Date) return Utilities.formatDate(v, TZ, p); return (v === null || v === undefined) ? '' : String(v); } function list_(name) { var s = sh_(name); if (s.getLastRow() < 1) return []; return s.getRange(1, 1, s.getLastRow(), 1).getValues() .map(function (r) { return String(r[0]).trim(); }) .filter(function (v) { return v !== ''; }) .filter(function (v, i) { return !(i === 0 && /^(model|part)$/i.test(v)); }); } function doGet(e) { try { var v = sh_('DATA').getDataRange().getValues(); var data = []; for (var i = 1; i < v.length; i++) { var r = v[i]; if (!r[0]) continue; data.push({ id: String(r[0]), tanggal: fmt_(r[1], 'dd/MM/yyyy'), nama: String(r[2]), line: String(r[3]), model: String(r[4]), part: String(r[5]), kurang: Number(r[6]) || 0, datang: Number(r[7]) || 0, sisa: Number(r[8]) || 0, statusAwal: String(r[9]), remarks: String(r[10]), updateCMC: String(r[11]), statusCMC: String(r[12] || 'Belum'), terakhir: fmt_(r[13], 'dd/MM/yyyy HH:mm'), updater: String(r[14] || '') }); } data.reverse(); // terbaru di atas return json_({ ok: true, data: data, models: list_('MODEL'), parts: list_('PART') }); } catch (err) { return json_({ ok: false, error: String(err) }); } } function doPost(e) { var lock = LockService.getScriptLock(); try { lock.waitLock(20000); var b = JSON.parse(e.postData.contents); if (b.action === 'add') return json_(add_(b)); if (b.action === 'update') return json_(upd_(b)); if (b.action === 'addModel') return json_(addItem_('MODEL', b.name, true)); if (b.action === 'addPart') return json_(addItem_('PART', b.name, false)); return json_({ ok: false, error: 'action tidak dikenal' }); } catch (err) { return json_({ ok: false, error: String(err) }); } finally { try { lock.releaseLock(); } catch (x) {} } } // Tambah nama baru ke kolom A tab MODEL / PART (tanpa duplikat, tidak peka huruf besar/kecil) function addItem_(tab, raw, upper) { var name = String(raw || '').replace(/\s+/g, ' ').trim(); if (upper) name = name.toUpperCase(); if (!name) throw new Error('Nama tidak boleh kosong'); if (name.length > 60) throw new Error('Nama terlalu panjang (maks 60 karakter)'); var s = sh_(tab); var list = list_(tab); for (var i = 0; i < list.length; i++) { if (list[i].toLowerCase() === name.toLowerCase()) return { ok: true, name: list[i], exists: true }; } var row = s.getLastRow() + 1; s.getRange(row, 1).setNumberFormat('@').setValue(name); // format teks: aman dari rumus return { ok: true, name: name, exists: false }; } function add_(b) { var items = (b.items && b.items.length) ? b.items : [b]; if (items.length > 30) throw new Error('Maksimal 30 part per kiriman'); var s = sh_('DATA'); var now = new Date(); var base = 'PP-' + Utilities.formatDate(now, TZ, 'ddMM-HHmmss'); var m = String(b.tanggal || '').match(/^(\d{4})-(\d{2})-(\d{2})$/); var tgl = m ? (m[3] + '/' + m[2] + '/' + m[1]) : Utilities.formatDate(now, TZ, 'dd/MM/yyyy'); var rows = [], ids = []; for (var i = 0; i < items.length; i++) { var it = items[i]; var kurang = Number(it.kurang); if (!String(it.part || '').trim()) throw new Error('Nama part kosong (part ke-' + (i + 1) + ')'); if (!isFinite(kurang) || kurang >= 0) throw new Error('Kekurangan harus angka negatif (part ke-' + (i + 1) + ')'); var id = i === 0 ? base : base + '-' + (i + 1); ids.push(id); rows.push([id, tgl, b.nama, b.line, b.model, it.part, kurang, 0, kurang, it.statusAwal, it.remarks || '', '', 'Belum', now]); } var row = s.getLastRow() + 1, n = rows.length; s.getRange(row, 2, n, 5).setNumberFormat('@'); // B-F teks (aman dari rumus) s.getRange(row, 10, n, 3).setNumberFormat('@'); // J-L teks s.getRange(row, 1, n, 14).setValues(rows); return { ok: true, id: ids[0], ids: ids }; } function upd_(b) { var s = sh_('DATA'); var last = s.getLastRow(); var ids = s.getRange(1, 1, last, 1).getValues(); var row = -1; for (var i = 1; i < ids.length; i++) { if (String(ids[i][0]) === String(b.id)) { row = i + 1; break; } } if (row < 0) throw new Error('ID tidak ditemukan: ' + b.id); var qty = Number(b.datang); if (!isFinite(qty) || qty <= 0) throw new Error('Qty datang harus lebih dari 0'); var r = s.getRange(row, 1, 1, 14).getValues()[0]; var datangBaru = (Number(r[7]) || 0) + qty; var sisa = (Number(r[6]) || 0) + datangBaru; var status = sisa >= 0 ? 'Complete' : (datangBaru > 0 ? 'Partial' : 'Belum'); s.getRange(row, 12).setNumberFormat('@'); s.getRange(row, 8, 1, 2).setValues([[datangBaru, sisa]]); // H, I s.getRange(row, 12, 1, 3).setValues([[b.updateCMC || '', status, new Date()]]); // L, M, N if (s.getMaxColumns() < 15) s.insertColumnsAfter(s.getMaxColumns(), 15 - s.getMaxColumns()); s.getRange(row, 15).setNumberFormat('@').setValue(String(b.updater || '').replace(/\s+/g, ' ').trim().slice(0, 60)); // O return { ok: true, datang: datangBaru, sisa: sisa, status: status }; }
